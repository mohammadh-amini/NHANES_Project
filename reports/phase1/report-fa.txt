\begin{abstract}
\noindent این مطالعه به بررسی ارتباط بین مصرف کل قند، وضعیت سیگار کشیدن و پوسیدگی دندان در بزرگسالان آمریکایی با استفاده از داده‌های پیمایش ملی سلامت و تغذیه (NHANES) ۲۰۲۱-۲۰۲۳ می‌پردازد. پس از ادغام مجموعه داده‌های دموگرافیک، رژیم غذایی، سلامت دهان و سیگار، نمونه‌ی تحلیلی نهایی شامل ۴,۷۸۸ بزرگسال ۱۸ سال و بالاتر به دست آمد. تحلیل‌های توصیفی نشان داد که ۳۷.۱۸ درصد از شرکت‌کنندگان پوسیدگی دندان داشتند و ۴۱.۳۳ درصد سابقه‌ی سیگار کشیدن داشتند. میانگین مصرف روزانه قند ۹۵.۴۷ گرم (انحراف معیار = ۶۸.۱۳) بود. توزیع مصرف قند دارای چولگی به راست بود (چولگی = ۲.۳۱) و مردان به طور قابل توجهی بیشتر از زنان قند مصرف می‌کردند. این یافته‌های اولیه نشان‌دهنده‌ی ارتباطات بالقوه‌ای است که در فازهای بعدی به طور رسمی آزمون خواهند شد.
\end{abstract}

\section{مقدمه}
پوسیدگی دندان یک نگرانی عمده‌ی بهداشت عمومی است که تحت تأثیر عوامل مختلف سبک زندگی و رژیم غذایی قرار دارد. هدف اصلی این پروژه پاسخ به این سؤال است: \textbf{آیا مصرف قند و سیگار با پوسیدگی دندان در بزرگسالان آمریکایی ارتباط دارند، حتی با کنترل سن و جنسیت؟} این گزارش یافته‌های اکتشافی و توصیفی اولیه‌ی این تحقیق را ارائه می‌کند.

\section{روش‌ها}
\subsection{منبع داده و طراحی مطالعه}
داده‌های این مطالعه از پیمایش ملی سلامت و تغذیه (NHANES)، برنامه‌ای که توسط مراکز کنترل و پیشگیری از بیماری‌ها (CDC) اجرا می‌شود، استخراج شده است. این یک \textbf{مطالعه مشاهده‌ای} است؛ بنابراین نمی‌توان روابط علت و معلولی را اثبات کرد.

\subsection{پردازش داده و جمعیت تحلیلی}
چهار مجموعه داده با استفاده از شناسه‌ی یکتای \texttt{SEQN} ادغام شدند: دموگرافیک (\texttt{DEMO\_L})، رژیم غذایی (\texttt{DR1TOT\_L})، سلامت دهان (\texttt{OHQ\_L}) و سیگار (\texttt{SMQ\_L}). پس از فیلتر افراد بالای ۱۸ سال و مدیریت داده‌های گمشده، نمونه‌ی نهایی شامل \textbf{۴,۷۸۸} نفر بود.

\subsection{ممیزی داده‌های گمشده}
قبل از پاکسازی، درصد داده‌های گمشده به شرح زیر بود: مصرف قند (۲۲.۸۲٪)، پوسیدگی دندان (۱۵.۷۸٪) و سیگار (۱۲.۰۰٪). تمام ردیف‌های دارای داده‌ی گمشده در متغیرهای اصلی از تحلیل حذف شدند.

\subsection{متغیرها}
متغیرهای اصلی مورد نظر عبارتند از:
\begin{itemize}
    \item \textbf{متغیر پاسخ:} پوسیدگی دندان (دودویی: بله/خیر)، مشتق شده از \texttt{OHQ660}.
    \item \textbf{متغیرهای توضیحی اصلی:} مصرف کل قند (\texttt{DR1TSUGR}) و وضعیت سیگار (\texttt{SMQ020}).
    \item \textbf{متغیرهای مخدوش‌گر:} سن (\texttt{RIDAGEYR}) و جنسیت (\texttt{RIAGENDR}).
    \item \textbf{وزن نمونه‌گیری:} \texttt{WTMEC2YR} برای تحلیل‌های وزن‌دار استفاده شد.
\end{itemize}

\subsection{فراوانی پوسیدگی دندان و سیگار}
\begin{itemize}
    \item \textbf{پوسیدگی دندان:} ۱,۷۸۰ شرکت‌کننده (۳۷.۱۸٪) پوسیدگی دندان داشتند، در حالی که ۳,۰۰۸ نفر (۶۲.۸۲٪) نداشتند.
    \item \textbf{سیگار:} ۱,۹۷۹ شرکت‌کننده (۴۱.۳۳٪) سابقه‌ی سیگار کشیدن (حداقل ۱۰۰ نخ) داشتند، در حالی که ۲,۸۰۹ نفر (۵۸.۶۷٪) نداشتند.
\end{itemize}

\section{بحث و نتیجه‌گیری}
آماده‌سازی داده‌ها و تحلیل توصیفی تأیید کرد که مجموعه داده‌های ادغام‌شده NHANES برای بررسی سؤال تحقیقاتی اصلی مناسب است. شواهد بصری و عددی اولیه حاکی از یک ارتباط بالقوه بین مصرف قند، سیگار و پوسیدگی دندان است که تفاوت‌های جنسیتی قابل توجهی در مصرف قند وجود دارد. در فازهای بعدی، استنباط آماری رسمی (آزمون فرض و مدل‌سازی رگرسیون) برای آزمون دقیق این ارتباط با کنترل متغیرهای مخدوش‌گر انجام خواهد شد.
